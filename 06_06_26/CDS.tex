		\end{equation*}

	\begin{frame}[t]{Les paramètres optimaux}
		\begin{equation*}
			\min_{a,b} \frac{1}{N}\sum_{i=1}^N \big( y_i - (ax_i+b) \big)^2
		\end{equation*}
		\begin{itemize}[<+->]
			\item On calcule $\bar x = \frac{1}{N}\sum_{i=1}^N x_i$ et $\bar y = \frac{1}{N}\sum_{i=1}^N y_i$
			\item On calcule ensuite
			\begin{equation*}
				a^\star = \frac{\sum_{i=1}^N (x_i-\bar x)(y_i-\bar y)}{\sum_{i=1}^N (x_i-\bar x)^2}\quad\text{et}\quad b^\star = \bar{y}-a^\star\bar x
			\end{equation*}
			\item La meilleure droite est alors $y=a^\star x + b^\star$
		\end{itemize}
	\end{frame}

	\subsection{Choix de la fonction de perte}
	
	
	\begin{frame}[t]{Réseaux de neurones : l'entraînement}
		\begin{itemize}[<+->]
			\item Il s'agit de trouver les "meilleurs" paramètres 
			\item Dans le cas de la régression linéaire, les meilleurs étaient ceux minimisant la MSE
		\end{itemize}
		\vspace{\baselineskip}
		\visible<3->{Avant d'entraîner un réseau de neurone, il faut :}
		\vspace{\baselineskip}
		\pause
		\begin{enumerate}[<+->]
			\item Définir ce que l'on attend de notre réseau
			\item Trouver une \textbf{fonction de perte} pertinente
		\end{enumerate}
	\end{frame}
	
	\begin{frame}{Exemples de fonctions de perte}
		\begin{center}
			\resizebox{\textwidth}{!}{
				\bgroup
				\everymath{\displaystyle}
				\renewcommand{\arraystretch}{2.2}
				
				\begin{tabular}{ccc}
					
					$ \frac{1}{n} \sum_{i=1}^{n} (y_i - \hat{y}_i)^2 $ & 
					$ \frac{1}{n} \sum_{i=1}^{n} |y_i - \hat{y}_i| $ & 
					$ \begin{cases} \frac{1}{2}(y-\hat{y})^2 & \text{si } |y-\hat{y}| \le \delta \\ \delta |y-\hat{y}| - \frac{1}{2}\delta^2 & \text{sinon} \end{cases} $ \\[4ex]
					
					$ - \frac{1}{n} \sum_{i=1}^{n} \left[ y_i \log(\hat{y}_i) + (1 - y_i) \log(1 - \hat{y}_i) \right] $ & 
					$ - \sum_{i=1}^{C} y_i \log(\hat{y}_i) $ & 
					$ -\alpha_t (1 - p_t)^\gamma \log(p_t) $ \\[4ex]
					
					$ \max(0, 1 - y \cdot \hat{y}) $ &
